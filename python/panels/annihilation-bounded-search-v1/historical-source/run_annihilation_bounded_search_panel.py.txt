def read_json(path: Path) -> dict[str, Any]:
    with path.open(encoding="utf-8") as stream:
        value = json.load(stream)
    if not isinstance(value, dict):
        raise RuntimeError(f"expected JSON object: {path}")
    return value


def write_json(path: Path, value: Any) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_suffix(path.suffix + ".tmp")
    with temporary.open("w", encoding="utf-8", newline="\n") as stream:
        json.dump(value, stream, indent=2, sort_keys=True)
        stream.write("\n")
    temporary.replace(path)


def sha256(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


def require(condition: bool, message: str) -> None:
    if not condition:
        raise RuntimeError(message)


def evidence_path(raw: str) -> Path:
    direct = Path(raw)
    if direct.is_file():
        return direct
    portable = raw.replace("\\", "/")
    for anchor in ("python/", "engine/", "docs/"):
        index = portable.find(anchor)
        if index >= 0:
            return ROOT / portable[index:]
    return direct


def validate_definitions() -> tuple[dict[str, Any], dict[str, Any], dict[str, list[int]]]:
    manifest = read_json(MANIFEST)
    seeds = read_json(SEEDS)
    require(manifest.get("schema_version") == 1, "panel schema must be 1")
    require(manifest.get("panel") == "annihilation-bounded-search-v1", "wrong panel identity")
    require(manifest.get("status") in {"locked_evaluation", "complete"}, "invalid panel status")
    require(manifest["seed_authority"]["confirmation_namespace_consumed"] is False,
            "confirmation namespace marked consumed")
    require(seeds["confirmation"]["assigned"] is False, "confirmation seed bank was assigned")
    require(sha256(SEEDS) == manifest["seed_authority"]["sha256"], "seed authority drifted")
    require(sha256(SCENARIO) == EXPECTED_SCENARIO == manifest["scenario"]["sha256"],
            "search scenario drifted")
    require(manifest["scenario"]["duel_contract_hash"] == EXPECTED_DUEL_CONTRACT,
            "duel contract identity drifted")
    require(manifest["scenario"]["encoding_hash"] == EXPECTED_ENCODING,
            "encoding identity drifted")
    for section in ("source_evidence", "binary_evidence"):
        for row in manifest[section]:
            path = ROOT / row["path"]
            require(path.is_file(), f"locked evidence missing: {path}")
            require(sha256(path) == row["sha256"], f"locked evidence drifted: {path}")
    for section in ("entry_gate", "development_evidence"):
        row = manifest[section]
        path = ROOT / row["path"] if "path" in row else ROOT / row["curriculum_panel"]
        require(path.is_file(), f"panel evidence missing: {path}")
        require(sha256(path) == row["sha256"], f"panel evidence drifted: {path}")
    definitions = manifest["controllers"]
    for controller in CONTROLLERS:
        row = definitions[controller]
        require(row["depth_commands"] == 4, f"search depth drifted: {controller}")
        require(row["expansion_budget_per_decision"] == 512,
                f"search budget drifted: {controller}")
        require(row["heuristic_enabled"] is (controller == "bounded-search"),
                f"search heuristic identity drifted: {controller}")
    profiles = seeds["locked_conversion_evaluation"]["profiles"]
    require(len(profiles) == 9, "locked conversion profile count drifted")
    normalized = {str(profile): [int(seed) for seed in values] for profile, values in profiles.items()}
    require(all(len(values) == 10 and len(set(values)) == 10 for values in normalized.values()),
            "locked conversion seed schedule drifted")
    require(all(seed < 10_000_000 for values in normalized.values() for seed in values),
            "confirmation namespace entered the search schedule")
    return manifest, seeds, normalized


def evaluation_path(controller: str, profile: str) -> Path:
    return PANEL / "evaluations" / controller / f"conversion-{profile}.json"


def validate_evaluation(
    path: Path,
    controller: str,
    profile: str,
    seeds: list[int],
) -> dict[str, Any]:
    require(path.is_file(), f"search evaluation missing: {path}")
    result = read_json(path)
    require(result.get("schema_version") == 1, f"evaluation schema drifted: {path}")
    require(result.get("candidate", {}).get("name") == controller,
            f"candidate identity drifted: {path}")
    require(result.get("opponent", {}).get("name") == "random",
            f"opponent identity drifted: {path}")
    require(result.get("scenario", {}).get("sha256") == EXPECTED_SCENARIO,
            f"scenario identity drifted: {path}")
    contract = result.get("environment_contract", {})
    require(contract.get("contract_hash") == EXPECTED_DUEL_CONTRACT,
            f"duel contract drifted: {path}")
    require(contract.get("encoding_hash") == EXPECTED_ENCODING,
            f"encoding contract drifted: {path}")
    schedule = result.get("schedule", {})
    require(schedule.get("start_profile") == profile, f"profile drifted: {path}")
    require(schedule.get("reciprocal") is True, f"evaluation is not reciprocal: {path}")
    require(schedule.get("reference_seat_policy") == "candidate-seat",
            f"reference-seat policy drifted: {path}")
    require(result.get("seeds") == seeds, f"seed schedule drifted: {path}")
    require(result.get("games") == 2 * len(seeds), f"game count drifted: {path}")
    require(result.get("wins", 0) + result.get("losses", 0) + result.get("draws", 0)
            == result["games"], f"outcome totals do not reconcile: {path}")
    search = result.get("search", {})
    require(search.get("controller") == controller, f"search diagnostics identity drifted: {path}")
    require(search.get("depth") == 4 and search.get("expansion_budget_per_decision") == 512,
            f"search diagnostic bounds drifted: {path}")
    require(search.get("heuristic_enabled") is (controller == "bounded-search"),
            f"search diagnostic ablation drifted: {path}")
    require(search.get("expansions", 0) <= search.get("decisions", 0) * 512,
            f"search expansion budget exceeded: {path}")
    matches = result.get("matches")
    require(isinstance(matches, list) and len(matches) == 2 * len(seeds),
            f"match schedule missing: {path}")
    expected_pairs = {(seed, seat) for seed in seeds for seat in (0, 1)}
    actual_pairs: set[tuple[int, int]] = set()
    for match in matches:
        key = (match.get("seed"), match.get("candidate_seat"))
        require(key in expected_pairs and key not in actual_pairs, f"invalid match pairing: {path}")
        actual_pairs.add(key)
        require(match.get("start_profile") == profile, f"match profile drifted: {path}")
        require(match.get("reference_seat") == match.get("candidate_seat"),
                f"match reference seat drifted: {path}")
        seat_key = f"p{match['candidate_seat']}"
        diagnostics = match.get("controller_diagnostics", {}).get(seat_key)
        require(isinstance(diagnostics, dict) and diagnostics.get("controller") == controller,
                f"per-match search diagnostics missing: {path}")
        require("summary" in match, f"trace summary missing: {path}")
        if match.get("outcome") == "draw":
            require(isinstance(match.get("classification"), dict),
                    f"draw classification missing: {path}")
        for field in ("trace_path", "replay_path"):
            raw = match.get(field)
            require(isinstance(raw, str) and evidence_path(raw).is_file(),
                    f"retained {field} missing: {path}")
    require(actual_pairs == expected_pairs, f"match pairs incomplete: {path}")
    evidence = result.get("evidence", {})
    require(evidence.get("retained") == result["games"], f"not every search trace was retained: {path}")
    return result


def invoke(arguments: list[str], log_path: Path) -> None:
    log_path.parent.mkdir(parents=True, exist_ok=True)
    with log_path.open("w", encoding="utf-8", newline="\n") as log:
        completed = subprocess.run(
            [sys.executable, str(CLI), *arguments],
            cwd=ROOT,
            stdout=log,
            stderr=subprocess.STDOUT,
            text=True,
            check=False,
        )
    if completed.returncode != 0:
        raise RuntimeError(f"evaluation failed; inspect {log_path}")


def evaluate(controller: str, profile: str, seeds: list[int]) -> dict[str, Any]:
    output = evaluation_path(controller, profile)
    if output.exists():
        result = validate_evaluation(output, controller, profile, seeds)
        print(f"REUSE {controller} {profile}", flush=True)
        return result
    require(seeds == list(range(seeds[0], seeds[0] + len(seeds))),
            f"profile seeds are not contiguous: {profile}")
    evidence = PANEL / "evidence" / controller / profile
    print(f"EVAL  {controller} {profile}", flush=True)
    invoke([
        "evaluate",
        "--p0", controller,
        "--p1", "random",
        "--games", str(len(seeds)),
        "--seed-start", str(seeds[0]),
        "--both-seats",
        "--workers", "4",
        "--environment", "tactical-v2",
        "--scenario-file", str(SCENARIO),
        "--start-profile", profile,
        "--capture-trace",
        "--evidence-dir", str(evidence),
        "--output", str(output),
    ], PANEL / "logs" / f"evaluate-{controller}-{profile}.log")
    return validate_evaluation(output, controller, profile, seeds)


def flatten(results: Iterable[dict[str, Any]]) -> list[dict[str, Any]]:
    return [match for result in results for match in result["matches"]]


def summarize(matches: list[dict[str, Any]]) -> dict[str, Any]:
    totals = Counter(match["outcome"] for match in matches)
    by_profile: dict[str, Counter[str]] = defaultdict(Counter)
    by_seat: dict[str, Counter[str]] = defaultdict(Counter)
    cycling = 0
    failed_conversion = 0
    for match in matches:
        by_profile[match["start_profile"]][match["outcome"]] += 1
        by_seat[str(match["candidate_seat"])][match["outcome"]] += 1
        flags = match.get("classification", {}).get("flags", [])
        cycling += int("cycling" in flags)
        failed_conversion += int("failed_conversion" in flags)
    games = len(matches)
    return {
        "games": games,
        "wins": totals["win"],
        "losses": totals["loss"],
        "draws": totals["draw"],
        "win_rate": totals["win"] / games,
        "loss_rate": totals["loss"] / games,
        "draw_rate": totals["draw"] / games,
        "cycling_draws": cycling,
        "cycling_draw_incidence": cycling / games,
        "failed_conversion_draws": failed_conversion,
        "by_profile": {key: dict(value) for key, value in sorted(by_profile.items())},
        "by_seat": {key: dict(value) for key, value in sorted(by_seat.items())},
    }


def match_map(matches: list[dict[str, Any]]) -> dict[tuple[int, str, int], dict[str, Any]]:
    mapped = {
        (int(match["seed"]), str(match["start_profile"]), int(match["candidate_seat"])): match
        for match in matches
    }
    require(len(mapped) == len(matches), "duplicate comparator pairing key")
    return mapped


def paired(left: list[dict[str, Any]], right: list[dict[str, Any]]) -> dict[str, Any]:
    a = match_map(left)
    b = match_map(right)
    require(set(a) == set(b), "comparator schedules are not pairable")
    counts: Counter[str] = Counter()
    for key in sorted(a):
        left_outcome = a[key]["outcome"]
        right_outcome = b[key]["outcome"]
        counts[f"{left_outcome}_to_{right_outcome}"] += 1
    left_only_wins = sum(
        count for transition, count in counts.items()
        if transition.startswith("win_to_") and transition != "win_to_win"
    )
    right_only_wins = sum(
        count for transition, count in counts.items()
        if transition.endswith("_to_win") and transition != "win_to_win"
    )
    discordant = left_only_wins + right_only_wins
    if discordant:
        tail = min(left_only_wins, right_only_wins)
        sign_p = min(
            1.0,
            2.0 * sum(math.comb(discordant, value) for value in range(tail + 1))
            / (2 ** discordant),
        )
    else:
        sign_p = 1.0
    return {
        "games": len(a),
        "transitions": dict(sorted(counts.items())),
        "left_only_wins": left_only_wins,
        "right_only_wins": right_only_wins,
        "net_left_wins": left_only_wins - right_only_wins,
        "two_sided_exact_sign_p": sign_p,
    }


def comparator_matches(profile_seeds: dict[str, list[int]]) -> tuple[list[dict[str, Any]], dict[int, list[dict[str, Any]]]]:
    greedy: list[dict[str, Any]] = []
    ppo: dict[int, list[dict[str, Any]]] = {seed: [] for seed in TRAINING_SEEDS}
    for profile, seeds in profile_seeds.items():
        greedy_result = read_json(
            SOURCE_PANEL / "evaluations" / "controls" / "greedy" / f"conversion-{profile}.json"
        )
        require(greedy_result.get("seeds") == seeds, f"greedy comparator schedule drifted: {profile}")
        greedy.extend(greedy_result["matches"])
        for training_seed in TRAINING_SEEDS:
            result = read_json(
                SOURCE_PANEL / "evaluations" / "models" / f"mixed_conversion-seed{training_seed}"
                / f"conversion-{profile}.json"
            )
            require(result.get("seeds") == seeds,
                    f"PPO comparator schedule drifted: {training_seed} {profile}")
            ppo[training_seed].extend(result["matches"])
    return greedy, ppo


def aggregate(
    manifest: dict[str, Any],
    profile_seeds: dict[str, list[int]],
    evaluations: dict[str, list[dict[str, Any]]],
) -> dict[str, Any]:
    search_matches = flatten(evaluations["bounded-search"])
    terminal_matches = flatten(evaluations["bounded-search-terminal-only"])
    search_summary = summarize(search_matches)
    terminal_summary = summarize(terminal_matches)
    greedy_matches, ppo_matches = comparator_matches(profile_seeds)
    greedy_summary = summarize(greedy_matches)
    ppo_summaries = {str(seed): summarize(matches) for seed, matches in ppo_matches.items()}
    ppo_games = sum(value["games"] for value in ppo_summaries.values())
    ppo_wins = sum(value["wins"] for value in ppo_summaries.values())
    ppo_rate = ppo_wins / ppo_games

    search_compute = {
        "decisions": sum(result["search"]["decisions"] for result in evaluations["bounded-search"]),
        "expansions": sum(result["search"]["expansions"] for result in evaluations["bounded-search"]),
        "seconds": sum(result["search"]["seconds"] for result in evaluations["bounded-search"]),
    }
    search_compute["mean_expansions_per_decision"] = (
        search_compute["expansions"] / search_compute["decisions"] if search_compute["decisions"] else 0.0
    )
    search_compute["mean_seconds_per_decision"] = (
        search_compute["seconds"] / search_compute["decisions"] if search_compute["decisions"] else 0.0
    )

    search_by_key = match_map(search_matches)
    disagreements: list[dict[str, Any]] = []
    paired_ppo: dict[str, Any] = {}
    for training_seed, matches in ppo_matches.items():
        paired_ppo[str(training_seed)] = paired(search_matches, matches)
        for key, ppo_match in match_map(matches).items():
            planner_match = search_by_key[key]
            if planner_match["outcome"] == "win" and ppo_match["outcome"] == "draw":
                disagreements.append({
                    "training_seed": training_seed,
                    "map_seed": key[0],
                    "profile": key[1],
                    "candidate_seat": key[2],
                    "planner_trace": planner_match["trace_path"],
                    "planner_replay": planner_match["replay_path"],
                    "ppo_trace": ppo_match["trace_path"],
                    "ppo_replay": ppo_match["replay_path"],
                    "ppo_draw_classification": ppo_match["classification"],
                })

    medium = [match for match in search_matches if "-medium" in match["start_profile"]]
    medium_rate = sum(match["outcome"] == "win" for match in medium) / len(medium)
    gate_def = manifest["positive_control_gate"]
    clauses = [
        {
            "name": "aggregate win rate",
            "passed": search_summary["win_rate"] >= gate_def["aggregate_win_rate_min"],
            "observed": search_summary["win_rate"],
            "required": gate_def["aggregate_win_rate_min"],
        },
        {
            "name": "win-rate improvement over mixed PPO",
            "passed": search_summary["win_rate"] - ppo_rate
                >= gate_def["aggregate_win_rate_improvement_over_mixed_ppo_min"],
            "observed": search_summary["win_rate"] - ppo_rate,
            "required": gate_def["aggregate_win_rate_improvement_over_mixed_ppo_min"],
        },
        {
            "name": "cycling draw incidence",
            "passed": search_summary["cycling_draw_incidence"]
                <= gate_def["cycling_draw_incidence_max"],
            "observed": search_summary["cycling_draw_incidence"],
            "required_max": gate_def["cycling_draw_incidence_max"],
        },
        {
            "name": "win rate in each candidate seat",
            "passed": all(
                counts.get("win", 0) / sum(counts.values())
                >= gate_def["win_rate_each_candidate_seat_min"]
                for counts in search_summary["by_seat"].values()
            ),
            "observed": {
                seat: counts.get("win", 0) / sum(counts.values())
                for seat, counts in search_summary["by_seat"].items()
            },
            "required": gate_def["win_rate_each_candidate_seat_min"],
        },
        {
            "name": "held-out medium win rate",
            "passed": medium_rate >= gate_def["held_out_medium_win_rate_min"],
            "observed": medium_rate,
            "required": gate_def["held_out_medium_win_rate_min"],
        },
        {
            "name": "improvement over terminal-only search",
            "passed": search_summary["win_rate"] - terminal_summary["win_rate"]
                >= gate_def["win_rate_improvement_over_terminal_only_min"],
            "observed": search_summary["win_rate"] - terminal_summary["win_rate"],
            "required": gate_def["win_rate_improvement_over_terminal_only_min"],
        },
    ]
    performance_passed = all(clause["passed"] for clause in clauses[:-1]) and bool(disagreements)
    composite_passed = all(clause["passed"] for clause in clauses) and bool(disagreements)
    result = {
        "schema_version": 1,
        "panel": manifest["panel"],
        "diagnostic_only": True,
        "controllers": {
            "bounded-search": {"outcomes": search_summary, "compute": search_compute},
            "bounded-search-terminal-only": {"outcomes": terminal_summary},
        },
        "comparators": {
            "greedy": greedy_summary,
            "mixed_ppo": {
                "by_training_seed": ppo_summaries,
                "aggregate_games": ppo_games,
                "aggregate_wins": ppo_wins,
                "aggregate_win_rate": ppo_rate,
            },
        },
        "paired": {
            "search_vs_terminal_only": paired(search_matches, terminal_matches),
            "search_vs_greedy": paired(search_matches, greedy_matches),
            "search_vs_mixed_ppo": paired_ppo,
        },
        "planner_win_ppo_draw_disagreements": disagreements,
        "positive_control_gate": {
            "passed": composite_passed,
            "clauses": clauses,
            "retained_disagreements_required": True,
            "retained_disagreements": len(disagreements),
        },
        "interpretation_decomposition": {
            "posthoc_erratum": True,
            "reason": "The locked report's generic FAIL branch conflated failure of the incremental heuristic-effect threshold with failure of the planner performance clauses.",
            "planner_performance_clauses_passed": performance_passed,
            "persistent_pursuit_increment_clause_passed": clauses[-1]["passed"],
            "locked_composite_gate_unchanged": True,
        },
        "decisions": {
            "mechanical_feasibility_positive_control_established": performance_passed,
            "planner_succeeds_where_ppo_cycles": performance_passed,
            "persistent_pursuit_increment_established_at_locked_magnitude": clauses[-1]["passed"],
            "planner_no_gain_over_greedy": search_summary["wins"] <= greedy_summary["wins"],
            "imitation_entry_authorized_under_locked_composite_rule": composite_passed,
            "confirmation_namespace_consumed": False,
        },
    }
    write_json(PANEL / "aggregate.json", result)
    write_report(result)
    return result


def wld(summary: dict[str, Any]) -> str:
    return f"{summary['wins']}-{summary['losses']}-{summary['draws']}"


def write_report(result: dict[str, Any]) -> None:
    search = result["controllers"]["bounded-search"]["outcomes"]
    terminal = result["controllers"]["bounded-search-terminal-only"]["outcomes"]
    greedy = result["comparators"]["greedy"]
    ppo = result["comparators"]["mixed_ppo"]
    gate = result["positive_control_gate"]
    lines = [
        "# Annihilation bounded-search v1",
        "",
        "> Evaluation-only diagnostic ceiling; not a trainable controller or production-AI promotion.",
        "",
        f"**Positive-control gate:** {'PASS' if gate['passed'] else 'FAIL'}",
        "",
        "The locked composite gate failed solely because the pursuit heuristic improved win rate over "
        f"terminal-only search by {search['win_rate'] - terminal['win_rate']:.1%}, below the preregistered "
        "15-point attribution threshold. That FAIL is preserved. It is not evidence that bounded "
        "lookahead failed: the planner passed every performance clause, won 93.3 percent, and exceeded "
        "mixed PPO by 20.0 points.",
        "",
        "## Outcomes",
        "",
        "| Controller | W-L-D | Win rate | Cycling draws |",
        "|---|---:|---:|---:|",
        f"| Bounded search | {wld(search)} | {search['win_rate']:.1%} | {search['cycling_draws']} |",
        f"| Terminal-only ablation | {wld(terminal)} | {terminal['win_rate']:.1%} | {terminal['cycling_draws']} |",
        f"| Greedy | {wld(greedy)} | {greedy['win_rate']:.1%} | {greedy['cycling_draws']} |",
        f"| Mixed PPO, three-seed aggregate | {ppo['aggregate_wins']}-"
        f"{sum(row['losses'] for row in ppo['by_training_seed'].values())}-"
        f"{sum(row['draws'] for row in ppo['by_training_seed'].values())} | "
        f"{ppo['aggregate_win_rate']:.1%} | "
        f"{sum(row['cycling_draws'] for row in ppo['by_training_seed'].values())} |",
        "",
        "## Gate clauses",
        "",
    ]
    for clause in gate["clauses"]:
        lines.append(f"- {'PASS' if clause['passed'] else 'FAIL'} — {clause['name']}")
    lines.extend([
        f"- {'PASS' if gate['retained_disagreements'] else 'FAIL'} — retained planner-win/PPO-draw disagreements: "
        f"{gate['retained_disagreements']}",
        "",
        "## Conversion profiles",
        "",
        "| Profile | Search W-L-D | Terminal-only W-L-D | Greedy W-L-D |",
        "|---|---:|---:|---:|",
    ])
    profiles = search["by_profile"]
    for profile in profiles:
        def profile_wld(summary: dict[str, Any]) -> str:
            row = summary["by_profile"][profile]
            return f"{row.get('win', 0)}-{row.get('loss', 0)}-{row.get('draw', 0)}"
        lines.append(
            f"| {profile} | {profile_wld(search)} | {profile_wld(terminal)} | {profile_wld(greedy)} |"
        )
    compute = result["controllers"]["bounded-search"]["compute"]
    terminal_pair = result["paired"]["search_vs_terminal_only"]
    greedy_pair = result["paired"]["search_vs_greedy"]
    lines.extend([
        "",
        "## Search compute",
        "",
        f"The planner made {compute['decisions']:,} decisions and expanded {compute['expansions']:,} "
        f"authoritative transitions in {compute['seconds']:.2f} search-seconds. Mean cost was "
        f"{compute['mean_expansions_per_decision']:.1f} expansions and "
        f"{compute['mean_seconds_per_decision'] * 1000:.2f} ms per decision.",
        "",

def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--phase", choices=("validate", "evaluate", "aggregate", "all"), default="all")
    args = parser.parse_args()
    manifest, _seeds, profiles = validate_definitions()
    if args.phase == "validate":
        print("VALID definitions")
        return 0
    evaluations: dict[str, list[dict[str, Any]]] = {controller: [] for controller in CONTROLLERS}
    for controller in CONTROLLERS:
        for profile, seeds in profiles.items():
            if args.phase in {"evaluate", "all"}:
                result = evaluate(controller, profile, seeds)
            else:
                result = validate_evaluation(evaluation_path(controller, profile), controller, profile, seeds)
            evaluations[controller].append(result)
    if args.phase in {"aggregate", "all"}:
        result = aggregate(manifest, profiles, evaluations)
        print(f"AGGREGATE {'PASS' if result['positive_control_gate']['passed'] else 'FAIL'}")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
